```python
def migrate_v2_to_v3(conn) -> None:
    """Add flashcards + device_favorites + device_review_progress + device_view_history tables."""
    conn.execute(text("""
        CREATE TABLE IF NOT EXISTS flashcards (
            id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
            knowledge_point_id UUID NOT NULL REFERENCES knowledge_points(id),
            card_type VARCHAR(20) NOT NULL,
            front_title TEXT NOT NULL,
            front_hint TEXT,
            back_answer TEXT NOT NULL,
            back_key_features TEXT[],
            back_example_bridge_id UUID REFERENCES bridges(id) ON DELETE SET NULL,
            created_at TIMESTAMPTZ DEFAULT now(),
            updated_at TIMESTAMPTZ DEFAULT now()
        )
    """))
    conn.execute(text("CREATE INDEX IF NOT EXISTS ix_flashcards_kp ON flashcards(knowledge_point_id)"))

    conn.execute(text("""
        CREATE TABLE IF NOT EXISTS device_favorites (
            id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
            device_id UUID NOT NULL,
            target_type VARCHAR(20) NOT NULL,
            target_id UUID NOT NULL,
            created_at TIMESTAMPTZ DEFAULT now(),
            UNIQUE(device_id, target_type, target_id)
        )
    """))
    conn.execute(text("CREATE INDEX IF NOT EXISTS ix_device_favorites_device ON device_favorites(device_id)"))

    conn.execute(text("""
        CREATE TABLE IF NOT EXISTS device_review_progress (
            id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
            device_id UUID NOT NULL,
            flashcard_id UUID NOT NULL REFERENCES flashcards(id) ON DELETE CASCADE,
            mastery VARCHAR(20) NOT NULL DEFAULT 'needs_review',
            last_reviewed_at TIMESTAMPTZ,
            interval INTEGER NOT NULL DEFAULT 1,

    set_schema_version(engine, 3, "Add flashcards + device tables", conn=conn)
```

Rewrite `get_bridge()` to:
1. Query `Bridge` by id (existing)
2. Query `Chunk` where `bridge_id = bridge_id` and `lvl1 IN (four mainlines)`
3. Group chunks by `LVL1_TO_SECTION[chunk.lvl1]`, build L0/L1 for each section
4. Build deduplicated `sources` list from chunks' `source_id` (join to `Source` table)
5. Populate `MobileBridgeDetail.sections` and `sources`

```python
@router.get("/{bridge_id}")
def get_bridge(bridge_id: UUID, db: Session = Depends(get_db)) -> MobileResponse[dict]:
    bridge = db.query(Bridge).filter(Bridge.id == bridge_id).first()
    if not bridge:
        raise HTTPException(404, detail="桥梁不存在")

    # Aggregate sections from chunks
    from app.models import Source
    chunks = (
        db.query(Chunk)
        .filter(Chunk.bridge_id == bridge_id, Chunk.lvl1.in_(LVL1_TO_SECTION.keys()))
        .order_by(Chunk.created_at)
        .all()
    )

    # Build source mapping: source_id → display index
    source_ids_ordered: list[UUID] = []
    source_id_set: set[UUID] = set()
    for c in chunks:
        if c.source_id and c.source_id not in source_id_set:
            source_ids_ordered.append(c.source_id)
            source_id_set.add(c.source_id)

    source_map: dict[UUID, int] = {sid: i + 1 for i, sid in enumerate(source_ids_ordered)}
    sources_data: list[dict] = []
    if source_ids_ordered:
        src_rows = db.query(Source).filter(Source.id.in_(source_ids_ordered)).all()
        src_by_id = {s.id: s for s in src_rows}
        for sid in source_ids_ordered:
            s = src_by_id.get(sid)
            sources_data.append({
                "id": source_map[sid],
                "title": s.title if s else None,
                "author": None,
                "type": s.type if s else None,
                "excerpt": None,
                "location": None,
            })

    # Build sections
    sections: dict[str, dict] = {
        key: {"l0": [], "l1": []} for key in ["structure", "geography", "society", "history"]
    }
    for i, c in enumerate(chunks):
        section_key = LVL1_TO_SECTION.get(c.lvl1)
        if not section_key:
            continue
        summary = c.section_title or (c.content[:80] + "…" if len(c.content) > 80 else c.content)
        citations = [source_map[c.source_id]] if c.source_id in source_map else []
        sections[section_key]["l0"].append({"id": str(c.id), "summary": summary, "order": len(sections[section_key]["l0"]) + 1})
        sections[section_key]["l1"].append({"id": str(c.id), "summary": summary, "detail": c.content, "citations": citations})

    detail = MobileBridgeDetail(
        id=bridge.id, name=bridge.name, aliases=bridge.aliases,
        bridge_type=bridge.bridge_type, era=bridge.era, region=bridge.region,
        latitude=bridge.latitude, longitude=bridge.longitude,
        sections=BridgeSections(**sections),
        sources=sources_data,
        sources_count=len(sources_data),
    )
    return MobileResponse(data=detail.model_dump())
```

```python
# app/api/mobile/v1/search.py
"""Mobile API: Unified search endpoint."""
import re

from fastapi import APIRouter, Depends, Query
from sqlalchemy import or_
from sqlalchemy.orm import Session

from app.api.mobile.v1.schemas.common import MobileMeta, MobilePaginatedResponse
from app.api.mobile.v1.schemas.search import SearchResultItem
from app.core.database import get_db
from app.models import Bridge, TaxonomyNode
from app.models.knowledge_point import KnowledgePoint


router = APIRouter()


def _escape_like(s: str) -> str:
    return s.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")


def _highlight(text: str, keyword: str) -> str:
    if not text or not keyword:
        return text or ""
    pattern = re.compile(re.escape(keyword), re.IGNORECASE)
    return pattern.sub(lambda m: f"<em>{m.group()}</em>", text)


@router.get("/search")
def search(
    q: str = Query(..., min_length=1),
    type: str = Query(default="all", pattern="^(all|bridge|knowledge_point)$"),
    bridge_type: str | None = None,
    era: str | None = None,
    region: str | None = None,
    line: str | None = None,
    category: str | None = None,
    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=20, ge=1, le=100),
    db: Session = Depends(get_db),
) -> MobilePaginatedResponse[dict]:
    escaped = _escape_like(q)
    pattern = f"%{escaped}%"
    results: list[dict] = []
    total = 0

    # Bridge search
    if type in ("all", "bridge"):
        bq = db.query(Bridge).filter(
            or_(Bridge.name.ilike(pattern), Bridge.aliases.ilike(pattern))
        )
        if bridge_type:
            types = [t.strip() for t in bridge_type.split(",") if t.strip()]
            if types:
                bq = bq.filter(Bridge.bridge_type.in_(types))
        if era:
            eras = [e.strip() for e in era.split(",") if e.strip()]
            if eras:
                bq = bq.filter(Bridge.era.in_(eras))
        if region:
            bq = bq.filter(Bridge.region.ilike(f"%{_escape_like(region)}%"))

        b_total = bq.count()
        total += b_total
        bridges = bq.order_by(Bridge.name).offset((page - 1) * page_size).limit(page_size).all()
        for b in bridges:
            hl = _highlight(b.name, q) or _highlight(b.aliases, q) or b.name
            results.append(SearchResultItem(
                type="bridge", id=b.id, name=b.name, aliases=b.aliases,
                highlight=hl, bridge_type=b.bridge_type, era=b.era,
                region=b.region, latitude=b.latitude, longitude=b.longitude,
            ).model_dump())

    # Knowledge point search
    if type in ("all", "knowledge_point"):
        kq = db.query(KnowledgePoint).filter(
            or_(KnowledgePoint.name.ilike(pattern), KnowledgePoint.aliases.ilike(pattern))
        )
        if line or category:
            kq = kq.join(TaxonomyNode, KnowledgePoint.taxonomy_node_id == TaxonomyNode.id)
            if line:
                kq = kq.filter(TaxonomyNode.lvl1 == line)
            if category:
                kq = kq.filter(TaxonomyNode.lvl2 == category)

        k_total = kq.count()
        total += k_total
        kps = kq.order_by(KnowledgePoint.name).offset((page - 1) * page_size).limit(page_size).all()

        # Batch lookup taxonomy for line names
        tax_ids = [kp.taxonomy_node_id for kp in kps if kp.taxonomy_node_id]
        tax_map: dict = {}
        if tax_ids:
            tax_nodes = db.query(TaxonomyNode).filter(TaxonomyNode.id.in_(tax_ids)).all()
            tax_map = {t.id: t for t in tax_nodes}

        for kp in kps:
            hl = _highlight(kp.name, q) or _highlight(kp.aliases, q) or kp.name
            tax = tax_map.get(kp.taxonomy_node_id)
            results.append(SearchResultItem(
                type="knowledge_point", id=kp.id, name=kp.name, aliases=kp.aliases,
                highlight=hl, line=tax.lvl1 if tax else None,
            ).model_dump())
```

```python
@router.get("/clusters")
def get_bridge_clusters(
    bounds_ne_lat: float = Query(...),
    bounds_ne_lng: float = Query(...),
    bounds_sw_lat: float = Query(...),
    bounds_sw_lng: float = Query(...),
    zoom: int = Query(..., ge=1, le=20),
    bridge_type: str | None = None,
    era: str | None = None,
    region: str | None = None,
    db: Session = Depends(get_db),
) -> MobileResponse[dict]:
    """Grid-based map clustering."""
    if bounds_sw_lat >= bounds_ne_lat:
        raise HTTPException(400, detail="bounds_sw_lat must be less than bounds_ne_lat")

    grid_size = 180.0 / (2 ** zoom)

    lat_bucket = func.floor(Bridge.latitude / grid_size)
    lng_bucket = func.floor(Bridge.longitude / grid_size)

    query = db.query(
        lat_bucket.label("lat_b"),
        lng_bucket.label("lng_b"),
        func.count().label("cnt"),
        func.avg(Bridge.latitude).label("center_lat"),
        func.avg(Bridge.longitude).label("center_lng"),
        func.min(Bridge.id).label("rep_id"),
    ).filter(
        Bridge.latitude.isnot(None),
        Bridge.latitude.between(bounds_sw_lat, bounds_ne_lat),
        Bridge.longitude.between(bounds_sw_lng, bounds_ne_lng),
    )

    if bridge_type:
        types = [t.strip() for t in bridge_type.split(",") if t.strip()]
        if types:
            query = query.filter(Bridge.bridge_type.in_(types))
    if era:
        eras = [e.strip() for e in era.split(",") if e.strip()]
        if eras:
            query = query.filter(Bridge.era.in_(eras))
    if region:
        query = query.filter(Bridge.region.ilike(f"%{_escape_like(region)}%"))

    rows = query.group_by("lat_b", "lng_b").all()

    # Batch fetch representative bridge names
    rep_ids = [row.rep_id for row in rows]
    rep_map: dict = {}
    if rep_ids:
        reps = db.query(Bridge.id, Bridge.name, Bridge.bridge_type, Bridge.latitude, Bridge.longitude).filter(
            Bridge.id.in_(rep_ids)
        ).all()
        rep_map = {r.id: r for r in reps}

    clusters: list[dict] = []
    points: list[dict] = []

    for row in rows:
        rep = rep_map.get(row.rep_id)
        if row.cnt == 1 and rep:
            points.append({
                "id": str(rep.id), "name": rep.name, "bridge_type": rep.bridge_type,
                "latitude": rep.latitude, "longitude": rep.longitude,
            })
        else:
            clusters.append({
                "latitude": float(row.center_lat),
                "longitude": float(row.center_lng),
                "count": row.cnt,
                "representative": {"id": str(rep.id), "name": rep.name} if rep else None,
            })

    return MobileResponse(data={"clusters": clusters, "points": points})
```

```python
def _resolve_targets(db: Session, favorites: list[DeviceFavorite]) -> dict[UUID, dict]:
    """Batch-resolve target names for a list of favorites."""
    grouped: dict[str, list[UUID]] = {}
    for f in favorites:
        grouped.setdefault(f.target_type, []).append(f.target_id)

    result: dict[UUID, dict] = {}
    for ttype, ids in grouped.items():
        model, name_field, summary_fn = TARGET_MODELS.get(ttype, (None, None, None))
        if model is None:
            continue
        rows = db.query(model).filter(model.id.in_(ids)).all()
        for row in rows:
            result[row.id] = {
                "target_name": getattr(row, name_field, None),
                "target_summary": summary_fn(row),
                "thumbnail": None,
            }
    return result
```

```python
import json
import logging

from app.providers import ZhipuProvider
from app.providers.base import ChatMessage

logger = logging.getLogger(__name__)

GENERATE_PROMPT_TEMPLATE = """你是桥梁知识专家。根据以下参考资料，为知识点「{name}」提炼 3-7 条核心要点。

要求：
1. 每条要点包含 summary（一句话，20-40字）和 explanation（展开解释，80-200字）
2. 每条要点标注引用的资料编号（citations），编号从1开始
3. 按逻辑递进排序

参考资料：
{evidence}

输出严格的 JSON 格式（不要输出任何其他内容）：
{{"key_points": [{{"summary": "...", "explanation": "...", "citations": [1, 2]}}]}}"""


@router.post("/{kp_id}/generate-key-points")
async def generate_key_points(
    kp_id: UUID,
    request: Request,
    db: Session = Depends(get_db),
    user: User = Depends(get_current_user),
    ip: str = Depends(get_client_ip),
):
    kp = db.query(KnowledgePoint).filter(KnowledgePoint.id == kp_id).first()
    if not kp:
        raise HTTPException(404, detail="Knowledge point not found")

    # Find related chunks via taxonomy
    chunks = _find_chunks_for_kp(db, kp)
    if not chunks:
        raise HTTPException(422, detail="No chunks found for this knowledge point's taxonomy")

    # Build evidence text
    evidence_parts = []
    for i, c in enumerate(chunks, 1):
        evidence_parts.append(f"[{i}] {c.content}")
    evidence_text = "\n\n".join(evidence_parts)

    # LLM call
    provider = ZhipuProvider()
    prompt = GENERATE_PROMPT_TEMPLATE.format(name=kp.name, evidence=evidence_text)
    response = await provider.chat([ChatMessage(role="user", content=prompt)])

    if not response.content:
        raise HTTPException(502, detail="LLM returned empty response")

    # Parse JSON
    try:
        json_start = response.content.find("{")
        json_end = response.content.rfind("}") + 1
        data = json.loads(response.content[json_start:json_end])
        key_points_data = data.get("key_points", [])
    except (json.JSONDecodeError, ValueError) as e:
        logger.error("Failed to parse LLM response: %s", response.content[:200])
        raise HTTPException(502, detail="Failed to parse LLM response")

    # Replace old key points
    db.query(KnowledgePointKeyPoint).filter(
        KnowledgePointKeyPoint.knowledge_point_id == kp_id
    ).delete()

    new_kps = []
    for i, kp_data in enumerate(key_points_data, 1):
        citations = kp_data.get("citations", [])
        chunk_ids = [chunks[c - 1].id for c in citations if 1 <= c <= len(chunks)]
        kp_item = KnowledgePointKeyPoint(
            knowledge_point_id=kp_id,
            summary=kp_data.get("summary", ""),
            explanation=kp_data.get("explanation"),
            order_index=i,
            chunk_ids=chunk_ids,
        )
        db.add(kp_item)
        new_kps.append({"summary": kp_item.summary, "order_index": i})

    AuditService.log(
        db, user.id, "generate_key_points", "knowledge_point",
        str(kp.id), {"count": len(new_kps)}, ip,
    )
    db.commit()

    return {"generated": len(new_kps), "key_points": new_kps}


def _find_chunks_for_kp(db: Session, kp: KnowledgePoint, limit: int = 20) -> list:
    """Find chunks matching the knowledge point's taxonomy, with lvl3→lvl2 fallback."""
    if not kp.taxonomy_node_id:
        return []

    tax = db.query(TaxonomyNode).filter(TaxonomyNode.id == kp.taxonomy_node_id).first()
    if not tax:
        return []

    # Try lvl3 match first
    if tax.lvl3:
        chunks = (
            db.query(Chunk)
            .filter(Chunk.lvl1 == tax.lvl1, Chunk.lvl2 == tax.lvl2, Chunk.lvl3 == tax.lvl3)
            .order_by(Chunk.created_at)
            .limit(limit)
            .all()
        )
        if len(chunks) >= 3:
            return chunks

    # Fallback to lvl2
    if tax.lvl2:
        return (
            db.query(Chunk)
            .filter(Chunk.lvl1 == tax.lvl1, Chunk.lvl2 == tax.lvl2)
            .order_by(Chunk.created_at)
            .limit(limit)
            .all()
        )

    return []
```
